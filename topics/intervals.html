Intervals

Merge overlapping intervals, sweep line, and scheduling problems.

Technique Sweep Line O(n log n)

Key Facts

  • Sort by start: O(n log n) — sorting is almost always the first step
  • Merge overlapping: O(n) scan after sorting
  • Overlap condition: intervals [a,b] and [c,d] overlap when c ≤ b
  • Sweep line: process events (start/end) in sorted order
  • Space: O(n) for output or event arrays

How Interval Problems Work

An interval is a pair [start, end] representing a range. Interval problems typically ask you to merge overlapping ranges, find gaps, count overlapping intervals at a point, or schedule tasks without conflicts.

The foundational insight is sort by start time. Once sorted, you only need to compare each interval with the most recently processed one. Two intervals [a,b] and [c,d] overlap when c ≤ b (the second starts before the first ends). Their merge is [min(a,c), max(b,d)].

For counting simultaneous overlaps (e.g. "how many meeting rooms do we need?") use a sweep line: create events for interval starts (+1) and ends (−1), sort all events by time, and track the running count. The maximum count reached is the answer.

For the "insert interval" pattern, split the existing intervals into three groups: those fully before the new one, those that overlap (merge into one), and those fully after. Handle the overlap group in a single pass.

  • Sort by start first — almost every interval problem needs this.
  • Overlap check: a.start ≤ b.end AND b.start ≤ a.end.
  • Sweep line: convert intervals to events (+1 start, −1 end) to count overlaps at any point.
  • Watch out for inclusive vs exclusive endpoints — [1,2] and [2,3] may or may not overlap depending on the problem.

# ── Merge Overlapping Intervals ───────────────────────────────
def merge(intervals):
    intervals.sort(key=lambda x: x[0])
    out = [intervals[0][:]]
    for s, e in intervals[1:]:
        if s <= out[-1][1]:              # overlaps
            out[-1][1] = max(out[-1][1], e)
        else:
            out.append([s, e])
    return out

# ── Insert Interval ───────────────────────────────────────────
def insert(intervals, newInterval):
    out = []
    for i, (s, e) in enumerate(intervals):
        if e < newInterval[0]:           # fully before
            out.append([s, e])
        elif s > newInterval[1]:         # fully after: flush new then rest
            out.append(newInterval)
            return out + intervals[i:]
        else:                            # overlapping: merge
            newInterval[0] = min(newInterval[0], s)
            newInterval[1] = max(newInterval[1], e)
    out.append(newInterval)
    return out

import heapq
def min_meeting_rooms(intervals):
    intervals.sort(key=lambda x: x[0])
    heap = []                            # end times of active meetings
    for s, e in intervals:
        if heap and heap[0] <= s:
            heapq.heapreplace(heap, e)   # reuse a freed room
        else:
            heapq.heappush(heap, e)      # need a new room
    return len(heap)

#include <stdio.h>
#include <stdlib.h>

typedef struct { int start, end; } Interval;

static int by_start(const void* a, const void* b) {
    return ((const Interval*)a)->start - ((const Interval*)b)->start;
}

// ── Merge Overlapping Intervals ───────────────────────────────
// Writes merged intervals to out[]; returns how many.
int merge(Interval iv[], int n, Interval out[]) {
    if (n == 0) return 0;
    qsort(iv, n, sizeof(Interval), by_start);
    int m = 0;
    out[m++] = iv[0];
    for (int i = 1; i < n; i++) {
        if (iv[i].start <= out[m - 1].end) {          // overlaps
            if (iv[i].end > out[m - 1].end) out[m - 1].end = iv[i].end;
        } else {
            out[m++] = iv[i];
        }
    }
    return m;
}

// ── Insert Interval (intervals sorted, non-overlapping) ────────
int insert(Interval iv[], int n, Interval nw, Interval out[]) {
    int m = 0, i = 0;
    while (i < n && iv[i].end < nw.start) out[m++] = iv[i++];   // fully before
    while (i < n && iv[i].start <= nw.end) {                     // overlapping: merge
        if (iv[i].start < nw.start) nw.start = iv[i].start;
        if (iv[i].end > nw.end) nw.end = iv[i].end;
        i++;
    }
    out[m++] = nw;
    while (i < n) out[m++] = iv[i++];                            // fully after
    return m;
}

// ── Meeting Rooms II (min rooms needed) — sweep line ──────────
static int cmp_int(const void* a, const void* b) {
    return *(const int*)a - *(const int*)b;
}
int min_meeting_rooms(Interval iv[], int n) {
    int* starts = malloc(n * sizeof(int));
    int* ends = malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) { starts[i] = iv[i].start; ends[i] = iv[i].end; }
    qsort(starts, n, sizeof(int), cmp_int);
    qsort(ends, n, sizeof(int), cmp_int);
    int rooms = 0, best = 0, e = 0;
    for (int s = 0; s < n; s++) {
        if (starts[s] >= ends[e]) e++;   // a meeting ended: reuse its room
        else rooms++;                    // need a new room
        if (rooms > best) best = rooms;
    }
    free(starts); free(ends);
    return best;
}

static void print_all(const char* label, Interval* iv, int n) {
    printf("%s", label);
    for (int i = 0; i < n; i++) printf(" [%d,%d]", iv[i].start, iv[i].end);
    printf("\n");
}

int main(void) {
    Interval a[] = {{1, 3}, {2, 6}, {8, 10}, {15, 18}}, out[8];
    print_all("merge:", out, merge(a, 4, out));
    Interval b[] = {{1, 3}, {6, 9}};
    print_all("insert [2,5]:", out, insert(b, 2, (Interval){2, 5}, out));
    Interval c[] = {{0, 30}, {5, 10}, {15, 20}};
    printf("meeting rooms: %d\n", min_meeting_rooms(c, 3));
    return 0;
}
